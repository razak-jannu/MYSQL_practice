2.ORDER BY clause *****

--1. Display all employees in ascending order of salary.
SELECT * FROM employees
ORDER BY salary ASC;
--2. Display all employees in descending order of salary.
SELECT * from employees
order by salary DESC;
--3. Display employee names and salaries in ascending order of salary.
SELECT emp_name,salary
from employees
ORDER BY salary ASC;
--4. Display all employees in alphabetical order of employee name.
SELECT * from employees
ORDER BY emp_name ASC;
--5. Display all employees in reverse alphabetical order of employee name.
select * from employees
ORDER BY emp_name DESC;
--6. Display all employees sorted by experience in ascending order.
SELECT * from employees
order by experience ASC;
--7. Display all employees sorted by experience in descending order.
SELECT * from employees
order by experience DESC;
--8. Display all IT department employees sorted by salary in descending order.
SELECT * from employees
WHERE department='it'
ORDER BY salary DESC;
--9. Display all Hyderabad employees sorted by employee name.
select * from employees
where city='hyderabad'
ORDER BY emp_name ASC;
--10. Display employee names, departments, and salaries sorted by department.
select emp_name,department,salary
from employees
ORDER BY department ASC;
--11. Display all employees sorted first by department and then by salary.
SELECT * from employees
ORDER BY department ASC, salary ASC;
--12. Display all employees sorted first by city and then by employee name.
SELECT * from employees
ORDER BY city asc , emp_name asc;
--13. Display all employees sorted by designation and then by experience in descending order.
select * from employees

--14. Display employees whose salary is greater than 50,000 sorted by salary in descending order.

WHERE salary >50000

--15. Display Finance department employees sorted by bonus in ascending order.

4. GROUP BY clause *****

-- 1. Find the number of employees in each department.
SELECT department,COUNT(*)
from employees
GROUP BY department;

-- 2. Find the total salary paid by each department.
SELECT department,SUM(salary)
FROM employees
GROUP BY department;

-- 3. Find the average salary of employees in each department.
SELECT department,AVG(salary) as avg_salary
FROM employees
group by department
ORDER BY avg_salary ASC;

-- 4. Find the minimum salary in each department.
SELECT department,min(salary)
FROM employees
GROUP BY department;

-- 5. Find the maximum salary in each department.
SELECT department,MAX(salary)
FROM employees
GROUP BY department;

-- 6. Find the total bonus given in each department.
SELECT department,SUM(bonus)
FROM employees
GROUP BY department;

-- 7. Find the average experience of employees in each department.
SELECT department,avg(experience)
FROM employees
GROUP BY department;

-- 8. Find the number of employees in each city.
SELECT city,COUNT(*)
FROM employees
GROUP BY city;

-- 9. Find the total salary paid to employees in each city.
SELECT city,FORMAT(sum(salary),2)
FROM employees
GROUP BY city;

-- 10. Find the average salary of employees in each city.
SELECT city,ROUND(AVG(salary),2)
FROM employees
GROUP BY city;

-- 11. Find the maximum salary in each city.
SELECT city,MAX(salary)
FROM employees
GROUP BY city;

-- 12. Find the minimum salary in each city.
SELECT city,min(salary)
FROM employees
GROUP BY city;

-- 13. Find the total bonus received by employees in each city.
SELECT city,SUM(bonus)
FROM employees
GROUP BY city;

-- 14. Find the average experience of employees in each city.
SELECT city,ROUND(AVG(experience),2) as avg_exp
FROM employees
GROUP BY city
ORDER BY avg_exp ASC;

-- 15. Find the number of employees in each designation.
SELECT designation, COUNT(*)
FROM employees
GROUP BY designation;

-- 16. Find the average salary for each designation.
SELECT designation,FORMAT(AVG(salary),2)
FROM employees
GROUP BY designation;

-- 17. Find the total salary for each designation.
SELECT designation,FORMAT(SUM(salary),2)
FROM employees
GROUP BY designation;

-- 18. Find the total salary and total bonus for each department.
SELECT department,sum(salary) as sum_salary ,SUM(bonus) as sum_bonus,
FORMAT(SUM(salary) + SUM(bonus),2) AS total_compensation
FROM employees
GROUP BY department;

-- 19. Find the number of employees and average salary for each department.
SELECT department,FORMAT(AVG(salary),2) as avg_salary,COUNT(*) as total_emp
FROM employees
GROUP BY department;

-- 20. Find the department-wise maximum salary and minimum salary.
SELECT department,MAX(salary) as max_salary,MIN(salary) as min_salary,
MAX(salary) - MIN(salary) AS salary_spread
FROM employees
GROUP BY department;


4. HAVING clause ****

-- 1. Find departments having more than 1 employee.
SELECT department, COUNT(*) as no_emp
FROM employees
GROUP BY department
HAVING no_emp >1;

-- 2. Find departments having more than 2 employees.
SELECT department, COUNT(*) as no_emp
FROM employees
GROUP BY department
HAVING no_emp >2;

-- 3. Find departments having less than 3 employees.
SELECT department,COUNT(*) as num_of_emp
FROM employees
GROUP BY department
HAVING num_of_emp <3;

-- 4. Find departments having exactly 2 employees.
SELECT department,COUNT(*) as num_of_emp
FROM employees
GROUP BY department
HAVING num_of_emp =2;

-- 5. Find departments where the total salary is greater than 100000.
SELECT department,sum(salary) as total_salary
FROM employees
GROUP BY department
HAVING total_salary > 100000;

-- 6. Find departments where the total salary is less than 150000.
SELECT department,sum(salary)
FROM employees
GROUP BY department
HAVING sum(salary) < 150000
ORDER BY sum(salary) ASC;
