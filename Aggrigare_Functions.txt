-- AGGREGATE FUNCTIONS – PRACTICE QUESTIONS

-- 1. Find the total number of employees.
SELECT COUNT(*)
FROM employees;
-- 2. Find the number of employees who work in the IT department.
SELECT count(*)
from employees
WHERE department ="it";
-- 3. Find the number of employees who have more than 5 years of experience.
SELECT COUNT(*)
FROM employees
WHERE experience >5;
-- 4. Find the total salary of all employees.
SELECT SUM(salary)
FROM employees; 

-- 5. Find the total bonus given to all employees.
SELECT SUM(bonus)
FROM employees;

-- 6. Find the total salary of employees working in the IT department.
SELECT sum(salary)
FROM employees
WHERE department='it';

-- 7. Find the average salary of all employees.
SELECT avg(salary)
FROM employees;

-- 8. Find the average experience of all employees.
SELECT avg(experience)
from employees;

-- 9. Find the average salary of employees working in the HR department.
SELECT avg(salary)
FROM employees 
WHERE department='hr';

-- 10. Find the lowest salary.
SELECT MIN(salary)
from employees;

-- 11. Find the lowest bonus.
SELECT MIN(bonus)
FROM employees;
-- 12. Find the employee with the lowest salary.
SELECT * FROM employees
WHERE salary=(SELECT MIN(salary) FROM employees);

-- 13. Find the highest salary.
SELECT MAX(salary)
from employees;

-- 14. Find the highest bonus.
SELECT MAX(bonus)
from employees;

-- 15. Find the employee with the highest salary.
SELECT * FROM employees
WHERE salary=(SELECT MAX(salary) from employees);
-- 16. Find the total salary of employees whose salary is greater than 50000.
SELECT SUM(salary)
FROM employees
WHERE salary > 50000;

-- 17. Find the average salary of employees from Hyderabad.
SELECT AVG(salary)
FROM employees
WHERE city='hyderabad';

-- 18. Find the maximum experience among employees in the IT department.
SELECT MAX(experience)
FROM employees
WHERE department='it';

-- 19. Find the minimum salary among employees in the Sales department.
SELECT MIN(salary)
FROM employees
WHERE department='sales';

-- 20. Find the total bonus of employees having more than 5 years of experience.
SELECT SUM(bonus)
FROM employees
WHERE experience >5;
